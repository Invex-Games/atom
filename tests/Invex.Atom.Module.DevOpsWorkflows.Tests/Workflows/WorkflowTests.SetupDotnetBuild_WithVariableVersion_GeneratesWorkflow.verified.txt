name: setup-dotnet-variable

trigger:
  branches:
    include: [ main ]

jobs:

  - job: SetupDotnetTarget
    displayName: SetupDotnetTarget
    pool:
      vmImage: ubuntu-latest
    steps:

      - checkout: self
        enabled: true
        fetchDepth: 0

      - task: UseDotNet@2
        displayName: Setup .NET $(DOTNET_VERSION)
        inputs:
          version: $(DOTNET_VERSION)

      - script: dotnet run --project AtomTest/AtomTest.csproj SetupDotnetTarget --skip --headless
        displayName: SetupDotnetTarget
        name: SetupDotnetTarget
